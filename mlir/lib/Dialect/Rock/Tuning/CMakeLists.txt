# Generate QuickTuningDb.inc which #includes every per-key .inc in sorted order.
# Filenames are the PascalCase rendering of the snake_case keys (e.g.
# Gfx942ConvF16.inc <-> gfx942_conv_f16), so sorting by filename yields
# an initializer in lexicographic key order -- a precondition for binary-search
# lookups.
set(QUICK_TUNING_DB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/QuickTuningDb)
file(GLOB _inc_files CONFIGURE_DEPENDS "${QUICK_TUNING_DB_DIR}/*.inc")
list(SORT _inc_files)

set(_content "// Auto-generated by ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt\n")
foreach(_f IN LISTS _inc_files)
  string(APPEND _content "#include \"${_f}\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QuickTuningDb.inc" "${_content}")

# QuickTuningDb.inc is generated into the build tree above, so add it to the
# private include path.
target_include_directories(MLIRRockTuning
  PRIVATE
  ${CMAKE_CURRENT_BINARY_DIR}
)
