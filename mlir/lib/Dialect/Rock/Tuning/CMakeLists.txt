# Generate QuickTuningDb.inc which #includes every per-key .inc in sorted order.
# Filenames are the PascalCase rendering of the snake_case keys (e.g.
# Gfx942ConvF16.inc <-> gfx942_conv_f16), so sorting by filename yields
# an initializer in lexicographic key order -- a precondition for binary-search
# lookups.
set(QUICK_TUNING_DB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/QuickTuningDb)
file(GLOB _inc_files CONFIGURE_DEPENDS "${QUICK_TUNING_DB_DIR}/*.inc")
list(SORT _inc_files)

set(_content "// Auto-generated by ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt\n")
foreach(_f IN LISTS _inc_files)
  string(APPEND _content "#include \"${_f}\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QuickTuningDb.inc" "${_content}")

# Generate SmartTuningDb.inc the same way: glue every per-(arch, op) model .inc
# in Models/ into one table, sorted by filename so the initializer is in
# lexicographic key order (a precondition for the binary-search resolution).
set(SMART_TUNING_DB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/Models)
file(GLOB _model_inc_files CONFIGURE_DEPENDS "${SMART_TUNING_DB_DIR}/*.inc")
list(SORT _model_inc_files)

set(_model_content "// Auto-generated by ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt\n")
foreach(_f IN LISTS _model_inc_files)
  string(APPEND _model_content "#include \"${_f}\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SmartTuningDb.inc" "${_model_content}")

# QuickTuningDb.inc / SmartTuningDb.inc are generated into the build tree above,
# so add it to the private include path.
target_include_directories(MLIRRockTuning
  PRIVATE
  ${CMAKE_CURRENT_BINARY_DIR}
)
